import { error } from '@sveltejs/kit';
import pages from '../../../reference/karento/src/lib/server/pages.json';
import { composeSitePage, resolveSiteRoute, siteRoutes } from '../../lib/server/site.mjs';
import { personalizePage } from '../../lib/server/dealer-profile.mjs';

const bodies = import.meta.glob('../../../reference/karento/src/lib/server/pages/*.html', {
  query: '?raw', import: 'default', eager: true
}) as Record<string, string>;

export function load({ params, url }: { params: { path?: string }; url: URL }) {
  const route = (params.path || '').replace(/\/$/, '');
  if (!Object.hasOwn(siteRoutes, route)) error(404, 'Page not found');
  const sourceKey = resolveSiteRoute(route);
  const page = (pages as Record<string, unknown>)[sourceKey];
  if (!page) error(404, 'Page not found');
  return { page: personalizePage(sourceKey, composeSitePage(sourceKey, page,
    bodies[`../../../reference/karento/src/lib/server/pages/${sourceKey}.html`],
    bodies['../../../reference/karento/src/lib/server/pages/index-2.html'],
    bodies['../../../reference/karento/src/lib/server/pages/index-3.html']), url) };
}
