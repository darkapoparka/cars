## src/routes/listing-detail-v1/[id]/+page.ts
1: import { listReturn } from '$data/journeys';
2: import { error } from '@sveltejs/kit';
3: import { featuredVehicles } from '$data/inventory';
4: import type { PageLoad } from './$types';
5: 
6: export const load: PageLoad = ({ params, url }) => {
7:   const isCanonicalId = /^[1-9]\d*$/.test(params.id);
8:   const vehicleId = Number(params.id);
9: 
10:   if (!isCanonicalId || !Number.isSafeInteger(vehicleId)) {
11:     error(404, 'Автомобилът не е намерен.');
12:   }
13: 
14:   const vehicle = featuredVehicles.find((item) => item.id === vehicleId);
15: 
16:   if (!vehicle) {
17:     error(404, 'Автомобилът не е намерен.');
18:   }
19: 
20:   const recommendations = featuredVehicles
21:     .filter((item) => item.id !== vehicle.id)
22:     .sort((a, b) => {
23:       const sameMakeA = a.make === vehicle.make ? 1 : 0;
24:       const sameMakeB = b.make === vehicle.make ? 1 : 0;
25:       const sameBodyA = a.body === vehicle.body ? 1 : 0;
26:       const sameBodyB = b.body === vehicle.body ? 1 : 0;
27:       return sameMakeB + sameBodyB - (sameMakeA + sameBodyA);
28:     })
29:     .slice(0, 3);
30: 
31:   return { returnTo: listReturn(url.searchParams.get('return'), '/listing-grid'), vehicle, recommendations };
32: };

1: import { listReturn } from '$data/journeys';
2: import { error } from '@sveltejs/kit';
3: import { blogCategories, blogPosts } from '$data/editorial';
4: import type { PageLoad } from './$types';
5: 
6: export const load: PageLoad = ({ params, url }) => {
7:   const isCanonicalId = /^[1-9]\d*$/.test(params.id);
8:   const postId = Number(params.id);
9: 
10:   if (!isCanonicalId || !Number.isSafeInteger(postId)) {
11:     error(404, 'Материалът не е намерен.');
12:   }
13: 
14:   const post = blogPosts.find((item) => item.id === postId);
15:   if (!post) {
16:     error(404, 'Материалът не е намерен.');
17:   }
18: 
19:   const related = blogPosts
20:     .filter((item) => item.id !== post.id)
21:     .sort((a, b) => Number(b.category === post.category) - Number(a.category === post.category))
22:     .slice(0, 3);
23: 
24:   const categories = blogCategories.map((category) => ({
25:     category,
26:     count: blogPosts.filter((item) => item.category === category).length
27:   }));
28: 
29:   const tags = [...new Set(blogPosts.map((item) => item.tag))];
30: 
31:   return { returnTo: listReturn(url.searchParams.get('return'), '/blog'), post, related, categories, tags };
32: };
