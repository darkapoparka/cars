## src/lib/components/vehicles/VehicleFinanceCalculator.svelte
1: <script lang="ts">
2:   import { vehicleContactHref } from '$data/journeys';
3:   import { resolve } from '$app/paths';
4:   import { formatVehiclePrice } from '$data/inventory';
5: 
6:   let { priceEur, vehicleId }: { priceEur: number; vehicleId: number } = $props();
7: 
8:   const financeTerms = [12, 24, 36, 48, 60] as const;
9:   let downPaymentEur = $state(0);
10:   let termMonths = $state<(typeof financeTerms)[number]>(60);
11: 
12:   let normalizedDownPayment = $derived(Math.min(Math.max(Number(downPaymentEur) || 0, 0), priceEur));
13:   let financedPrincipal = $derived(Math.max(priceEur - normalizedDownPayment, 0));
14:   let principalPerMonth = $derived(Math.round(financedPrincipal / termMonths));
15: 
16:   function normalizeDownPayment() {
17:     downPaymentEur = normalizedDownPayment;
18:   }
19: </script>
20: 
21: <div class="dn-finance-calculator">
22:   <header>
23:     <h2>Калкулатор за финансиране</h2>
24:     <p>Променете първоначалната вноска и срока, за да видите оставащата главница.</p>
25:   </header>
26: 
27:   <div class="dn-finance-calculator__fields">
28:     <label>
29:       <span>Първоначална вноска</span>
30:       <span class="dn-finance-calculator__input">
31:         <input
32:           type="number"
33:           min="0"
34:           max={priceEur}
35:           step="500"
36:           bind:value={downPaymentEur}
37:           onblur={normalizeDownPayment}
38:           aria-describedby="finance-disclaimer"
39:         />
40:         <b>€</b>
41:       </span>
42:     </label>
43: 
44:     <label>
45:       <span>Срок</span>
46:       <select bind:value={termMonths} aria-describedby="finance-disclaimer">
47:         {#each financeTerms as term (term)}
48:           <option value={term}>{term} месеца</option>
49:         {/each}
50:       </select>
51:     </label>
52:   </div>
53: 
54:   <dl class="dn-finance-calculator__result" aria-live="polite">
55:     <div>
56:       <dt>Оставаща главница</dt>
57:       <dd>{formatVehiclePrice(financedPrincipal)}</dd>
58:     </div>
59:     <div>
60:       <dt>Главница / месец</dt>
61:       <dd>{formatVehiclePrice(principalPerMonth)}</dd>
62:     </div>
63:   </dl>
64: 
65:   <p id="finance-disclaimer" class="dn-finance-calculator__disclaimer">

67:   </p>
68:

70:     Обсъдете финансиране

72: </div>
