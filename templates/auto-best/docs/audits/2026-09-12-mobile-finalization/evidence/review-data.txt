226: const isBlogCategory = (value: string | null): value is BlogCategory =>
227:   Boolean(value && blogCategories.includes(value as BlogCategory));

229: export const parseBlogFilters = (params: URLSearchParams): BlogFilters => ({
230:   q: params.get('q')?.trim() ?? '',
231:   category: isBlogCategory(params.get('category')) ? params.get('category') as BlogCategory : ''
232: });
233: 
234: const normalize = (value: string) => value.trim().toLocaleLowerCase('bg-BG');
235: 
236: export const filterBlogPosts = (posts: BlogPost[], filters: BlogFilters) => {
237:   const query = normalize(filters.q);
238:   return posts.filter((post) => {
239:     if (filters.category && post.category !== filters.category) return false;
240:     if (!query) return true;
241:     return normalize(`${post.title} ${post.text} ${post.category} ${post.tag}`).includes(query);
242:   });
243: };
244: 
245: export const blogFilterHref = (filters: BlogFilters, category: BlogCategory | '') => {
246:   const params = new URLSearchParams();
247:   if (filters.q) params.set('q', filters.q);
248:   if (category) params.set('category', category);
249:   const query = params.toString();
250:   return query ? `/blog?${query}` : '/blog';
251: };
